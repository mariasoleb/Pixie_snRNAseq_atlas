# Merge All Files

## Goal

Merge all annotated tissue-level Seurat objects into a single multi-organ atlas, re-normalize with SCTransform, and recompute a joint UMAP embedding.

prepare_for_merge <- function(rds_file) {
  message("Processing: ", rds_file)

  obj <- readRDS(rds_file)

  ## 1️⃣ Set default assay to RNA
  if (!"RNA" %in% names(obj@assays)) {
    stop("RNA assay not found in: ", rds_file)
  }
  DefaultAssay(obj) <- "RNA"

  ## 2️⃣ Remove SCT assay if present
  if ("SCT" %in% names(obj@assays)) {
    obj[["SCT"]] <- NULL
  }

  ## 3️⃣ Join layers in RNA assay (Seurat v5 requirement)
  obj[["RNA"]] <- JoinLayers(obj[["RNA"]])

  ## 4️⃣ Sanity checks (recommended)
  stopifnot(
    ncol(obj) > 0,
    nrow(obj) > 0,
    "counts" %in% names(obj@assays$RNA@layers)
  )

  return(obj)
}
