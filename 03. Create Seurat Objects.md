# Create Seurat Objects

## Goal

Load simpleaf quantification output into Seurat objects, apply quality control filters (mitochondrial and chloroplast content, feature count thresholds), and merge replicates per tissue.

```r
#run for each rep and tissue
library(SingleCellExperiment)
library(fishpond)
library(Seurat)
# define count matrix format
custom_format <- list("counts" = c("U", "S", "A"),
                      "spliced" = c("S"),
                      "unspliced" = c("U"),
                      "ambiguous" = c("A"))

# Load count matrix
sce <- fishpond::loadFry("alevin/STEM2_simpleaf/af_quant",
                         outputFormat = custom_format)

# Create Seurat object
seurat_obj <- CreateSeuratObject(counts(sce), min.cells = 3, min.features = 100)

# Define output path
output_path <- "./alevin_seurat/STEM2_simpleaf.rds"

# Create the directory if it doesn't exist
dir.create(dirname(output_path), recursive = TRUE, showWarnings = FALSE)

# Save the Seurat object
saveRDS(seurat_obj, file = output_path)

# Print message confirming save
cat("Seurat object saved at:", output_path, "\n")


## quality control


seurat_obj<-readRDS("./alevin_seurat/STEM1_simpleaf.rds")


counts_matrix <- GetAssayData(seurat_obj, layer = "counts")
# Filter genes expressed in at least 3 cells
genes_to_keep <- rowSums(counts_matrix > 0) >= 3
seurat_obj <- subset(seurat_obj, features = names(genes_to_keep[genes_to_keep]))


####filter high mito, chloro and Enecator nuclei #####
seurat_obj[["percent.mt"]] <- PercentageFeatureSet(seurat_obj, pattern = "ViviM")
seurat_obj[["percent.chl"]] <- PercentageFeatureSet(seurat_obj, pattern = "ViviC")


seurat_obj$percent.mt[is.nan(seurat_obj$percent.mt)] <- 0
seurat_obj$percent.chl[is.nan(seurat_obj$percent.chl)] <- 0



pdf("plots_alevin/Features_VLN_STEM1.pdf", width = 20, height = 5)
  VlnPlot(seurat_obj, features = c("nFeature_RNA", "nCount_RNA",
                      "percent.mt", "percent.chl"),
                 ncol = 4,pt.size = 0 )
dev.off()

  ####filter cells####
  seurat_obj <- subset(seurat_obj, subset = percent.mt < 10)
  seurat_obj <- subset(seurat_obj, subset = percent.chl < 10)


  seurat_obj <- subset(seurat_obj, subset = nFeature_RNA > 200 & nFeature_RNA < 3000)


  pdf("plots_alevin/Features_VLN_STEM1_afterclean.pdf", width = 20, height = 5)
  VlnPlot(seurat_obj, features = c("nFeature_RNA", "nCount_RNA",
                      "percent.mt", "percent.chl"),
                 ncol = 4,pt.size = 0 )
dev.off()

  #merge replicates into one seurat
library(Seurat)

# Load the two Seurat objects
bud1 <- readRDS("./alevin_seurat_clean/BUD1_clean.rds")
bud1[['Tissue_Rep']]<-"BUD_1"
bud2 <- readRDS("./alevin_seurat_clean/BUD2_clean.rds")
bud2[['Tissue_Rep']]<-"BUD_2"
# Merge them into one object
merged_obj_bud <- merge(
  bud1,
  y = bud2,
  add.cell.ids = c("BUD1", "BUD2"), # prefix cell barcodes
  project = "BUD_MERGED"
)

# Save merged object
saveRDS(merged_obj_bud, file = "./alevin_seurat_clean/BUD_merged_seurat.rds")
```
