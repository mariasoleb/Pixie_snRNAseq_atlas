# Quantify Using Simpleaf

## Goal

Quantify snRNA-seq reads across all tissue samples using simpleaf, and generate per-sample quality control reports using alevinQC.

```bash
# Define filename pattern
reads1_pat="_R1_001_preprocessed"
reads2_pat="_R2_"


reads1="$(find -L /your_wd/NEWLEAF2 -name "*$reads1_pat*" -type f | sort | awk -v OFS=, '{$1=$1;print}' | paste -sd, -)"

reads2="$(find -L /your_wd/NEWLEAF2 -name "*$reads2_pat*" -type f | sort | awk -v OFS=, '{$1=$1;print}' | paste -sd, -)"


  nohup simpleaf quant \
--reads1 $reads1 \
--reads2 $reads2 \
--threads 8 \
--index $IDX_DIR/index \
--chemistry GEXSCOPE-V2-preprocessed \
--resolution cr-like \
--knee \
--anndata-out \
--output $AF_SAMPLE_DIR/NEWLEAF2_simpleaf &




# loop thorugh all samples (2 lanes)

for sample_dir in "BUD1" "BUD2" "FLOWER1" "FLOWER2" "NEWLEAF1" "NEWLEAF2" "OLDLEAF1" "OLDLEAF2" "STEM1" "STEM2" "ROOT1" "ROOT2"; do
    sample_name=$(basename "$sample_dir")  # e.g. Sample1

    # Get all matching R1 and R2 files, comma-separated
    reads1=$(find "/your_wd/$sample_dir" -name "*_R1_001_preprocessed*.fastq.gz" | sort | paste -sd "," -)
    reads2=$(find "/your_wd/$sample_dir" -name "*_R2_001.fastq.gz" | sort | paste -sd "," -)

echo "simpleaf quant --reads1 \"$reads1\" --reads2 \"$reads2\" --threads 8 --index \"$IDX_DIR/index\" --knee --chemistry GEXSCOPE-V2-preprocessed --resolution cr-like --anndata-out --output \"$AF_SAMPLE_DIR/${sample_name}_simpleaf\""
done > commands.sh

nohup parallel -j 1 :::: commands.sh > simpleafMap.log 2>&1 &
```

```r
#Quality control

library(alevinQC)

simpleafQCReport(
  simpleafQuantDir="./NEWLEAF2_simpleaf",
  sampleId="NEWLEAF2",
  outputFile="simpleafQC_NEWLEAF2.html",
  outputDir = "./NEWLEAF2_simpleaf",
  outputFormat = "html_document",
  showCode = FALSE,
  forceOverwrite = FALSE,
  knitrProgress = FALSE,
  quiet = FALSE,
  ignorePandoc = FALSE,
  customCBList = list()
)
```
